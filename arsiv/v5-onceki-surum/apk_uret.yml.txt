          enum TrackCategory { fx, sticker, subtitle, pip, mainVideo, voiceover, music }

          class StudioClip {
            String id;
            String title;
            TrackCategory track;
            int startMs;
            int durationMs;
            bool isSelected;
            bool isStabilized;
            bool isOpticalFlow;
            bool isAiUpscaled;
            bool isVideoDenoised;
            bool isAudioDenoised;
            bool isMirrored;
            int rotationDeg;
            double speed;
            double brightness;
            double contrast;
            double saturation;
            List<int> keyframesMs;

            StudioClip({
              required this.id,
              required this.title,
              required this.track,
              required this.startMs,
              required this.durationMs,
              this.isSelected = false,
              this.isStabilized = false,
              this.isOpticalFlow = true,
              this.isAiUpscaled = false,
              this.isVideoDenoised = false,
              this.isAudioDenoised = true,
              this.isMirrored = false,
              this.rotationDeg = 0,
              this.speed = 1.0,
              this.brightness = 1.0,
              this.contrast = 1.0,
              this.saturation = 1.0,
              List<int>? keyframesMs,
            }) : keyframesMs = keyframesMs ?? [];

            int get endMs => startMs + durationMs;

            StudioClip cloneWith({int? newStartMs, int? newDurationMs}) {
              return StudioClip(
                id: 'c_${DateTime.now().microsecondsSinceEpoch}',
                title: title,
                track: track,
                startMs: newStartMs ?? startMs,
                durationMs: newDurationMs ?? durationMs,
                isStabilized: isStabilized,
                isOpticalFlow: isOpticalFlow,
                isAiUpscaled: isAiUpscaled,
                isVideoDenoised: isVideoDenoised,
                isAudioDenoised: isAudioDenoised,
                isMirrored: isMirrored,
                rotationDeg: rotationDeg,
                speed: speed,
                brightness: brightness,
                contrast: contrast,
                saturation: saturation,
              );
            }
          }

            void _toast(String msg) {
              ScaffoldMessenger.of(context).clearSnackBars();
              ScaffoldMessenger.of(context).showSnackBar(
                SnackBar(
                  content: Text(msg, style: const TextStyle(fontSize: 11, color: Colors.white)),
                  backgroundColor: const Color(0xFF27272F),
                  duration: const Duration(milliseconds: 1300),
                  behavior: SnackBarBehavior.floating,

            void _splitSelected() {
              final sel = selectedClip;
              if (sel == null || sel.durationMs < 1000) return;
              setState(() {
                final half = sel.durationMs ~/ 2;
                sel.durationMs = half;
                clips.add(sel.cloneWith(newStartMs: sel.startMs + half, newDurationMs: half));
              });
              _toast('Klip ikiye bölündü (Ctrl+B)');
            }

            void _razorAll() {
              setState(() {
                final List<StudioClip> added = [];
                for (final c in clips) {
                  if (c.durationMs >= 1600) {
                    final half = c.durationMs ~/ 2;
                    c.durationMs = half;
                    added.add(c.cloneWith(newStartMs: c.startMs + half, newDurationMs: half));
                  }
                }
                clips.addAll(added);
              });
              _toast('Tüm katmanlar aynı anda bölündü (Ctrl+Shift+K)');
            }

            void _trimSelected(bool head) {
              final sel = selectedClip;
              if (sel == null || sel.durationMs <= 1000) return;
              setState(() {
                if (head) sel.startMs += 500;
                sel.durationMs -= 500;
              });
              _toast(head ? 'Başı kırpıldı (Q)' : 'Sonu kırpıldı (W)');
            }

            void _deleteSelected(bool ripple) {
              final sel = selectedClip;
              if (sel == null) return;
              setState(() {
                final st = sel.startMs;
                final dur = sel.durationMs;
                final trk = sel.track;
                clips.removeWhere((c) => c.id == sel.id);
                if (ripple) {
                  for (final c in clips.where((x) => x.track == trk && x.startMs > st)) {
                    c.startMs = math.max(0, c.startMs - dur);
                  }
                }
                if (clips.isNotEmpty) clips.first.isSelected = true;
              });
              _toast(ripple ? 'Kaydırarak silindi (Ripple)' : 'Boşluklu silindi (Lift)');
            }

            void _duplicateSelected() {
              final sel = selectedClip;
              if (sel == null) return;
              setState(() {
                clips.add(sel.cloneWith(newStartMs: sel.endMs));
              });
              _toast('Klip çoğaltıldı');
            }

            void _detachAudio() {
              final sel = selectedClip;
              if (sel == null) return;
              setState(() {
                clips.add(StudioClip(
                  id: 'vo_${DateTime.now().microsecondsSinceEpoch}',
                  title: '🔉 ${sel.title} (Ayrılan Ses)',
                  track: TrackCategory.voiceover,
                  startMs: sel.startMs,
                  durationMs: sel.durationMs,
                ));
              });
              _toast('Ses bağımsız kanala ayrıldı');
            }
